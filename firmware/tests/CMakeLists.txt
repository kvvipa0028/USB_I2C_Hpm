cmake_minimum_required(VERSION 3.16)
project(bridge_native_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -Werror -Wconversion -Wshadow -pedantic -UNDEBUG)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include_directories(../common ../usb_i2c/src)
add_executable(test_common test_common.c ../common/i2c_timing.c ../common/i2c_table.c ../common/i2c_recovery.c)
add_executable(test_protocol test_protocol.c ../usb_i2c/src/bridge_protocol.c)
add_executable(test_port test_port.c ../usb_i2c/src/i2c_port.c
    ../common/i2c_timing.c ../common/i2c_table.c ../common/i2c_recovery.c)
target_include_directories(test_port BEFORE PRIVATE stubs)
add_executable(test_usb_slot test_usb_slot.c ../usb_i2c/src/usb_i2c.c)
target_include_directories(test_usb_slot BEFORE PRIVATE stubs)
add_test(NAME common COMMAND test_common)
add_test(NAME protocol COMMAND test_protocol)
add_test(NAME port_state_machine COMMAND test_port)
add_test(NAME usb_slot COMMAND test_usb_slot)

cmake_minimum_required(VERSION 3.16)
project(siggen_native_tests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
add_compile_options(-Wall -Wextra -Werror -Wconversion -Wshadow -pedantic -UNDEBUG)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
include_directories(../src)
add_executable(test_command test_command.c ../src/siggen_command.c)
add_executable(test_peer test_peer.c ../src/siggen_peer.c ../src/siggen_session.c ../src/siggen_command.c)
add_executable(test_cdc_buffer test_cdc_buffer.c ../src/cdc_buffer.c)
add_executable(test_cdc_backend test_cdc_backend.c ../src/cdc_acm.c ../src/cdc_buffer.c)
add_executable(test_recovery_gpio test_recovery_gpio.c ../src/siggen_recovery_gpio.c ../src/siggen_peer.c ../src/siggen_command.c)
target_include_directories(test_cdc_backend BEFORE PRIVATE stubs)
target_include_directories(test_recovery_gpio BEFORE PRIVATE stubs)
add_test(NAME command COMMAND test_command)
add_test(NAME peer COMMAND test_peer)
add_test(NAME cdc_buffer COMMAND test_cdc_buffer)
add_test(NAME cdc_backend COMMAND test_cdc_backend)
add_test(NAME recovery_gpio COMMAND test_recovery_gpio)
